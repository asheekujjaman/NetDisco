#NetDisco Install in Ubuntu Server

#Setp: 1
#Update Ubuntu Server First, update the Ubuntu package list and upgrade installed packages.
sudo apt update
sudo apt upgrade -y

#If needed, reboot the server:
sudo reboot


#Step: 2
#Install Required Packages:
sudo apt install -y build-essential curl cpanminus libdbd-pg-perl libsnmp-perl libssl-dev libio-socket-ssl-perl python3 python3-venv postgresql postgresql-contrib libcrypt-des-perl snmp ufw


#Step: 3
#Create Netdisco Linux User. Create a dedicated Linux user for running Netdisco.
sudo adduser --disabled-password --gecos "" netdisco


#Step: 4
#Configure PostgreSQL Database
#Switch to PostgreSQL User
sudo su - postgres

#Create PostgreSQL User for Netdisco
createuser -DRSP netdisco
#note:You will be asked to enter a password. Save this password because it will be needed later in the Netdisco configuration file.

#Create Netdisco Database
createdb -O netdisco netdisco

#Exit PostgreSQL User
exit


#Step: 5
#Configure PostgreSQL Authentication. Open the PostgreSQL authentication configuration file:
sudo nano /etc/postgresql/14/main/pg_hba.conf

#Add this line before other local authentication rules:
local   netdisco        netdisco                                md5
#Save and exit: CTRL + O; Enter; CTRL + X
#Restart PostgreSQL:
sudo systemctl restart postgresql
#Test database login:
psql -U netdisco -d netdisco -h localhost
#If login is successful, exit PostgreSQL:
\q


#Step: 6
#Install Netdisco via CPAN. Switch to the Netdisco user:
sudo su - netdisco
#Install Netdisco locally inside the Netdisco user’s home directory:
curl -L https://cpanmin.us/ | perl - --notest --local-lib ~/perl5 App::Netdisco
#This command downloads and installs Netdisco and its required Perl dependencies. and take long time 10-20 minuets. if failed any dependencies install that manually than try. search in AI 
#TacacsPlus this often failed. If fail in your process than try this by a normal user exit nesdisco user. sudo apt install -y libauthen-tacacsplus-perl.
#Test the module: perl -MAuthen::TacacsPlus -e 'print "Authen::TacacsPlus OK\n"'
#if show okay. than run installation again.


#Step: 7
#Create Netdisco Command Links. Create the bin directory and link Netdisco commands:
mkdir -p ~/bin
ln -sf ~/perl5/bin/{localenv,netdisco-*} ~/bin/

#Test Netdisco backend status:
~/bin/netdisco-backend status
#Example output: Netdisco Backend                                          [Not Running]
#This is normal. It means Netdisco is installed, but the backend service is not running yet.


#Step: 8
#Create Netdisco Configuration File
#Create the environment directory:
mkdir -p ~/environments
#Copy the default deployment configuration:
cp ~/perl5/lib/perl5/auto/share/dist/App-Netdisco/environments/deployment.yml ~/environments/
#Set secure permission:
chmod 600 ~/environments/deployment.yml
#Edit the configuration file:
nano ~/environments/deployment.yml
#Update the database section:
database:
  name: 'netdisco'
  user: 'netdisco'
  pass: 'YOUR_DATABASE_PASSWORD'
  host: 'localhost'
#Replace YOUR_DATABASE_PASSWORD with the PostgreSQL password created earlier.
#Update the SNMP community section:
community:
  - 'public'


#Step: 9
#Deploy Netdisco
#Run the Netdisco deployment command:
~/bin/netdisco-deploy
#During deployment, Netdisco will create database tables and ask to create an initial admin user for web. than ask some question. answer yes for all to default location.


#Step: 10
#Start Netdisco Services. 
#Start Netdisco web service:
~/bin/netdisco-web start
#Start Netdisco backend service:
~/bin/netdisco-backend start
#Check status:
~/bin/netdisco-web status
~/bin/netdisco-backend status
#Expected output:
#Netdisco Web        [Running]
#Netdisco Backend    [Running]

#Configure Firewall
#Exit from the Netdisco user:
exit
#Allow SSH and Netdisco web port:
sudo ufw allow 22/tcp
sudo ufw allow 5000/tcp
sudo ufw enable
sudo ufw status

#Access Netdisco Web Interface
http://SERVER_IP:5000

#If any error occurs in any step than take help from AI for related error.

#After this you start post installation process.
